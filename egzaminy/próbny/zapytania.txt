CREATE VIEW Dane_Parkowania AS
SELECT parkowanie.Nr_rejestracyjny,
       samochody.marka,
       parkowanie.Data,
       parkowanie.godzina_rozp,
       parkowanie.czas
  FROM parkowanie INNER JOIN samochody ON parkowanie.Nr_rejestracyjny = samochody.Nr_rejestracyjny

SELECT strefy.Nazwa,
       COUNT(parkowanie.id_strefy)
  FROM strefy INNER JOIN parkowanie ON strefy.Id_strefy = parkowanie.id_strefy
GROUP BY strefy.Nazwa
HAVING COUNT(parkowanie.id_strefy) < 14
